cmake_minimum_required(VERSION 3.21)
project(staged_reset_consumers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(configqml 0.5.1 REQUIRED COMPONENTS Qml)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Qml)
add_executable(old-native old_native.cpp)
target_link_libraries(old-native PRIVATE configqml::core)
# This process imports the installed module without linking configqml.
add_executable(import-only import_only.cpp)
target_link_libraries(import-only PRIVATE Qt6::Core Qt6::Qml)
add_executable(abi-layout abi_layout.cpp)
target_link_libraries(abi-layout PRIVATE configqml::configqml)
option(STAGED_RESET_NEW "Build the additive API consumer" OFF)
if(STAGED_RESET_NEW)
    add_executable(new-native new_native.cpp)
    target_link_libraries(new-native PRIVATE configqml::core)
endif()
