cmake_minimum_required(VERSION 3.21)
project(overlay_preserve_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(configqml 0.5.1 REQUIRED COMPONENTS Core Qml)
find_package(Qt6 REQUIRED COMPONENTS Test Qml)
add_executable(overlay-preserve ../test_overlay_preserve.cpp)
target_include_directories(overlay-preserve PRIVATE ..)
target_link_libraries(overlay-preserve PRIVATE configqml::core Qt6::Test)
add_executable(layout layout.cpp)
target_link_libraries(layout PRIVATE configqml::configqml)
# This driver deliberately has no configqml link or source-tree import path.
add_executable(import-only import.cpp)
target_link_libraries(import-only PRIVATE Qt6::Qml)
