cmake_minimum_required(VERSION 3.21)
project(configqml_components LANGUAGES CXX)
if(MODE STREQUAL "optional")
    find_package(configqml REQUIRED COMPONENTS Core OPTIONAL_COMPONENTS Qml)
elseif(MODE STREQUAL "required")
    find_package(configqml REQUIRED COMPONENTS Qml)
elseif(MODE STREQUAL "default")
    find_package(configqml REQUIRED)
else()
    message(FATAL_ERROR "Unknown component test mode: ${MODE}")
endif()
if(NOT configqml_FOUND OR NOT configqml_Core_FOUND OR NOT TARGET configqml::core)
    message(FATAL_ERROR "Core discovery failed")
endif()
if(CMAKE_DISABLE_FIND_PACKAGE_Qt6Qml)
    if(configqml_Qml_FOUND OR TARGET configqml::configqml)
        message(FATAL_ERROR "Unavailable optional Qml was imported")
    endif()
else()
    if(NOT configqml_Qml_FOUND OR NOT TARGET configqml::configqml)
        message(FATAL_ERROR "Available Qml was not imported")
    endif()
endif()
