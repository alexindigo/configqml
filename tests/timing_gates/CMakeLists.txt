cmake_minimum_required(VERSION 3.21)
project(timing_gate_probe LANGUAGES CXX)
add_library(work STATIC work.cpp)
add_executable(timing_probe main.cpp)
target_link_libraries(timing_probe PRIVATE work)
target_include_directories(timing_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_definitions(timing_probe PRIVATE EXPECTED_TIMING=${EXPECTED_TIMING})
if(SANITIZER_SOURCE STREQUAL "target")
    target_compile_options(work PRIVATE -fsanitize=undefined)
    target_link_options(timing_probe PRIVATE -fsanitize=undefined)
elseif(SANITIZER_SOURCE STREQUAL "interface")
    target_compile_options(work INTERFACE -fsanitize=undefined)
    target_link_options(timing_probe PRIVATE -fsanitize=undefined)
elseif(SANITIZER_SOURCE STREQUAL "expression")
    target_compile_options(work PRIVATE "$<$<CONFIG:Release>:-fsanitize=undefined>")
    target_link_options(timing_probe PRIVATE -fsanitize=undefined)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/ConfigQmlTimingGates.cmake")
cmake_language(EVAL CODE "cmake_language(DEFER DIRECTORY \"${CMAKE_SOURCE_DIR}\" CALL configqml_configure_timing_gates \"${CMAKE_CURRENT_SOURCE_DIR}\" timing_probe)")
