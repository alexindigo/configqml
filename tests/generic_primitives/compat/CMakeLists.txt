cmake_minimum_required(VERSION 3.21)
project(configqml_old_clients LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(configqml 0.5.3 REQUIRED COMPONENTS Qml)
find_package(Qt6 6.12 REQUIRED COMPONENTS Qml)
add_executable(old-native native.cpp)
target_link_libraries(old-native PRIVATE configqml::core)
add_executable(old-metaobject metaobject.cpp)
target_link_libraries(old-metaobject PRIVATE configqml::configqml)
# A genuinely import-only client, with no ConfigQml link or source headers.
add_executable(old-qml qml.cpp)
target_link_libraries(old-qml PRIVATE Qt6::Qml)
if(CONFIGQML_LAYOUT_SOURCE)
    add_executable(public-layouts "${CONFIGQML_LAYOUT_SOURCE}")
    target_link_libraries(public-layouts PRIVATE configqml::configqml)
    target_compile_options(public-layouts PRIVATE -Wno-invalid-offset)
endif()
